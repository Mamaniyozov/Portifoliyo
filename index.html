from django.http import JsonResponse
from django.views.decorators.http import require_GET
from .models import Developer

@require_GET
def get_profile(request):
    # Fetch profile data with prefetched skills
    developer = Developer.objects.select_related('profile').first()
    
    response_data = {
        "name": developer.full_name,
        "role": developer.role,
        "experience": developer.experience_years,
        "status": "Available for work (remote/office)",
        "active_projects": list(developer.projects.values('title', 'tech_stack'))
    }
    
    return JsonResponse(response_data, safe=False)
from django.db import models

class Developer(models.Model):
    full_name = models.CharField(max_length=200)
    role = models.CharField(max_length=100)
    experience_years = models.DecimalField(max_digits=3, decimal_places=1)
    email = models.EmailField(unique=True)
    is_available = models.BooleanField(default=True)

    def __str__(self):
        return self.full_name

class Project(models.Model):
    developer = models.ForeignKey(Developer, on_delete=models.CASCADE, related_name='projects')
    title = models.CharField(max_length=200)
    description = models.TextField()
    tech_stack = models.JSONField()
    github_link = models.URLField(blank=True)
